# Record the actual dependency sources, including FetchContent source overrides.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package-licenses.txt" "")
foreach(entry "SDL|SDL|LICENSE.txt" "ImGui|imgui|LICENSE.txt" "Abseil|abseil-cpp|LICENSE"
              "Tracy|tracy|LICENSE" "xxHash|xxhash|LICENSE")
  string(REPLACE "|" ";" fields "${entry}")
  list(GET fields 0 notice)
  list(GET fields 1 dependency)
  list(GET fields 2 filename)
  FetchContent_GetProperties(${dependency} SOURCE_DIR dependency_source)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/package-licenses.txt" "${notice}.txt\t${dependency_source}/${filename}\n")
endforeach()
